const allLetters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ".split('');

//渲染侧边字母栏
function renderSide(hasMap){
  sideIndexDom.innerHTML = '';
  allLetters.forEach(l=>{
    const div = document.createElement('div');
    div.className = 'side-item' + (hasMap[l]?' active':'');
    div.innerText = l;
    div.onclick = ()=>{
      const el = document.getElementById('g‑'+l);
      if(el) el.scrollIntoView({behavior:'smooth'});
    }
    sideIndexDom.appendChild(div);
  })
}

//渲染列表+搜索过滤
function renderList(keyword){
  let data = [...originData];
  if(keyword.trim()){
    const kw = keyword.trim().toUpperCase();
    data = data.filter(row=>{
      return row.姓名.includes(keyword) || row.姓氏大写字母 === kw;
    })
  }
  const groupMap = {};
  data.forEach(row=>{
    const l = row.姓氏大写字母;
    if(!groupMap[l]) groupMap[l]=[];
    groupMap[l].push(row);
  })
  const keys = Object.keys(groupMap).sort();
  const hasMap = {};
  keys.forEach(k=>hasMap[k]=true);
  renderSide(hasMap);

  listWrap.innerHTML = '';
  keys.forEach(letter=>{
    const arr = groupMap[letter];
    const groupDiv = document.createElement('div');
    groupDiv.id = 'g‑'+letter;
    const title = document.createElement('div');
    title.className = 'group-title';
    title.innerText = letter;
    groupDiv.appendChild(title);

    arr.forEach(item=>{
      const row = document.createElement('div');
      row.className = 'item-row';
      row.innerHTML = `
        <div class="name">${item.姓名}</div>
        <div class="money">¥${item.金额}</div>
        <div class="page">页码：${item.礼金本原始页码}</div>
      `;
      groupDiv.appendChild(row);
    })
    listWrap.appendChild(groupDiv);
  })

  const sumMoney = data.reduce((s,i)=>s+Number(i.金额),0);
  totalCountDom.innerText = data.length;
  totalMoneyDom.innerText = sumMoney;
}

keywordInput.addEventListener('input',e=>{
  renderList(e.target.value);
})
